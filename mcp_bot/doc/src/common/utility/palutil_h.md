# palutil.h

## Призначення файлу
Файл `palutil.h` визначає функції та структури для роботи з палітрами кольорів, перетворення кольорів та спеціальними кольоровими картами.

## Короткий опис
Цей файл містить набір утилітарних функцій для роботи з кольорами та палітрами, включаючи пошук найближчого кольору в палітрі, перетворення між кольоровими просторами RGB та HSV, створення таблиць перетворення кольорів, роботу зі спеціальними кольоровими картами тощо.

## Структура файлу
```cpp
#pragma once

#include <stdint.h>
#include "zstring.h"
#include "palentry.h"

struct FScriptPosition;
class FScanner;

// Функції для роботи з кольорами та палітрами
int BestColor(const uint32_t* pal, int r, int g, int b, int first = 1, int num = 255, const uint8_t* indexmap = nullptr);
int PTM_BestColor(const uint32_t* pal_in, int r, int g, int b, bool reverselookup, float powtable, int first = 1, int num = 255);
void DoBlending(const PalEntry* from, PalEntry* to, int count, int r, int g, int b, int a);
void MakeRemap(uint32_t* BaseColors, const uint32_t* colors, uint8_t* remap, const uint8_t* useful, int numcolors);
void MakeGoodRemap(uint32_t* BaseColors, uint8_t* Remap, const uint8_t* cmapdata = nullptr);

// Функції для перетворення кольорових просторів
void RGBtoHSV(float r, float g, float b, float *h, float *s, float *v);
void HSVtoRGB(float *r, float *g, float *b, float h, float s, float v);

// Функції для отримання кольорів з рядків
int V_GetColorFromString(const char* colorstring, FScriptPosition* sc = nullptr);
FString V_GetColorStringByName(const char* name, FScriptPosition* sc = nullptr);
int V_GetColor(const char* str, FScriptPosition* sc = nullptr);
int V_GetColor(FScanner& sc);
PalEntry averageColor(const uint32_t* data, int size, int maxout);

// Константи для спеціальних кольорових карт
enum
{
    NOFIXEDCOLORMAP = -1,
    INVERSECOLORMAP,
    GOLDCOLORMAP,
    REDCOLORMAP,
    GREENCOLORMAP,
    BLUECOLORMAP,
    REALINVERSECOLORMAP,
};

// Структури для спеціальних кольорових карт
struct FSpecialColormapParameters
{
    float Start[3], End[3];
};

struct FSpecialColormap
{
    float ColorizeStart[3];
    float ColorizeEnd[3];
    uint8_t Colormap[256];
    PalEntry GrayscaleToColor[256];
};

// Глобальні змінні
extern TArray<FSpecialColormap> SpecialColormaps;
extern uint8_t DesaturateColormap[31][256];

// Функції для роботи зі спеціальними кольоровими картами
int AddSpecialColormap(PalEntry *pe, float r1, float g1, float b1, float r2, float g2, float b2);
void InitSpecialColormaps(PalEntry* pe);
void UpdateSpecialColormap(PalEntry* BaseColors, unsigned int index, float r1, float g1, float b1, float r2, float g2, float b2);
int ReadPalette(int lumpnum, uint8_t* buffer);

// Перелік для маніпуляцій з кольорами
enum EColorManipulation
{
    CM_PLAIN2D = -2,
    CM_INVALID = -1,
    CM_DEFAULT = 0,
    CM_FIRSTSPECIALCOLORMAP,
};

#define CM_MAXCOLORMAP int(CM_FIRSTSPECIALCOLORMAP + SpecialColormaps.Size())
```

## Основні сутності

### Функції для роботи з кольорами та палітрами

#### BestColor
Функція для пошуку найближчого кольору в палітрі.

##### Призначення
Знаходить індекс кольору в палітрі, який найближче відповідає заданому RGB-кольору.

##### Параметри
- **pal**: Вказівник на палітру кольорів.
- **r, g, b**: Компоненти RGB-кольору.
- **first**: Перший індекс для пошуку (за замовчуванням 1).
- **num**: Кількість кольорів для пошуку (за замовчуванням 255).
- **indexmap**: Вказівник на карту індексів (за замовчуванням nullptr).

#### PTM_BestColor
Функція для пошуку найближчого кольору в палітрі з використанням степеневої таблиці.

##### Призначення
Знаходить індекс кольору в палітрі, який найближче відповідає заданому RGB-кольору, з використанням степеневої таблиці для більш точного відповідності сприйняттю людини.

##### Параметри
- **pal_in**: Вказівник на палітру кольорів.
- **r, g, b**: Компоненти RGB-кольору.
- **reverselookup**: Прапорець для зворотного пошуку.
- **powtable**: Степенева таблиця.
- **first**: Перший індекс для пошуку (за замовчуванням 1).
- **num**: Кількість кольорів для пошуку (за замовчуванням 255).

#### DoBlending
Функція для змішування кольорів.

##### Призначення
Змішує кольори з масиву `from` з кольорами `r, g, b, a` і зберігає результат у масиві `to`.

##### Параметри
- **from**: Вказівник на вихідні кольори.
- **to**: Вказівник на результуючі кольори.
- **count**: Кількість кольорів для змішування.
- **r, g, b, a**: Компоненти кольору для змішування.

#### MakeRemap
Функція для створення таблиці перетворення кольорів.

##### Призначення
Створює таблицю перетворення кольорів з масиву `colors` в масив `BaseColors`.

##### Параметри
- **BaseColors**: Вказівник на базові кольори.
- **colors**: Вказівник на кольори для перетворення.
- **remap**: Вказівник на таблицю перетворення.
- **useful**: Вказівник на масив корисних кольорів.
- **numcolors**: Кількість кольорів.

#### MakeGoodRemap
Функція для створення якісної таблиці перетворення кольорів.

##### Призначення
Створює якісну таблицю перетворення кольорів для базових кольорів.

##### Параметри
- **BaseColors**: Вказівник на базові кольори.
- **Remap**: Вказівник на таблицю перетворення.
- **cmapdata**: Вказівник на дані кольорової карти (за замовчуванням nullptr).

### Функції для перетворення кольорових просторів

#### RGBtoHSV
Функція для перетворення кольору з RGB в HSV.

##### Призначення
Перетворює колір з кольорового простору RGB в кольоровий простір HSV (відтінок, насиченість, яскравість).

##### Параметри
- **r, g, b**: Вхідні компоненти RGB-кольору.
- **h, s, v**: Вказівники на вихідні компоненти HSV-кольору.

#### HSVtoRGB
Функція для перетворення кольору з HSV в RGB.

##### Призначення
Перетворює колір з кольорового простору HSV в кольоровий простір RGB.

##### Параметри
- **r, g, b**: Вказівники на вихідні компоненти RGB-кольору.
- **h, s, v**: Вхідні компоненти HSV-кольору.

### Функції для отримання кольорів з рядків

#### V_GetColorFromString
Функція для отримання кольору з рядка.

##### Призначення
Перетворює рядок у форматі "rr gg bb" в індекс кольору.

##### Параметри
- **colorstring**: Рядок з кольором.
- **sc**: Вказівник на позицію в скрипті (за замовчуванням nullptr).

#### V_GetColorStringByName
Функція для отримання рядка кольору за його назвою.

##### Призначення
Шукає колір за його назвою в таблиці X11R6RGB і повертає рядок, придатний для V_GetColorFromString.

##### Параметри
- **name**: Назва кольору.
- **sc**: Вказівник на позицію в скрипті (за замовчуванням nullptr).

#### V_GetColor
Функція для отримання кольору за його назвою або рядком.

##### Призначення
Спочатку намагається отримати колір за його назвою, потім за рядком.

##### Параметри
- **str**: Назва або рядок кольору.
- **sc**: Вказівник на позицію в скрипті (за замовчуванням nullptr).

#### averageColor
Функція для обчислення середнього кольору.

##### Призначення
Обчислює середній колір з масиву даних.

##### Параметри
- **data**: Вказівник на дані кольорів.
- **size**: Розмір даних.
- **maxout**: Максимальне вихідне значення.

### Структури для спеціальних кольорових карт

#### FSpecialColormapParameters
Структура для параметрів спеціальної кольорової карти.

##### Призначення
Зберігає початкові та кінцеві значення для створення спеціальної кольорової карти.

##### Поля
- **Start**: Початкові значення для трьох компонентів кольору.
- **End**: Кінцеві значення для трьох компонентів кольору.

#### FSpecialColormap
Структура для спеціальної кольорової карти.

##### Призначення
Зберігає дані спеціальної кольорової карти, включаючи початкові та кінцеві значення для колоризації, таблицю перетворення кольорів та таблицю перетворення відтінків сірого в кольори.

##### Поля
- **ColorizeStart**: Початкові значення для колоризації.
- **ColorizeEnd**: Кінцеві значення для колоризації.
- **Colormap**: Таблиця перетворення кольорів.
- **GrayscaleToColor**: Таблиця перетворення відтінків сірого в кольори.

### Глобальні змінні

#### SpecialColormaps
Масив спеціальних кольорових карт.

##### Призначення
Зберігає всі спеціальні кольорові карти, які використовуються в грі.

#### DesaturateColormap
Таблиця для десатурації кольорів.

##### Призначення
Зберігає таблиці для десатурації кольорів з різними рівнями десатурації.

### Функції для роботи зі спеціальними кольоровими картами

#### AddSpecialColormap
Функція для додавання спеціальної кольорової карти.

##### Призначення
Додає нову спеціальну кольорову карту з заданими параметрами.

##### Параметри
- **pe**: Вказівник на палітру кольорів.
- **r1, g1, b1**: Початкові значення для колоризації.
- **r2, g2, b2**: Кінцеві значення для колоризації.

#### InitSpecialColormaps
Функція для ініціалізації спеціальних кольорових карт.

##### Призначення
Ініціалізує всі спеціальні кольорові карти з заданою палітрою.

##### Параметри
- **pe**: Вказівник на палітру кольорів.

#### UpdateSpecialColormap
Функція для оновлення спеціальної кольорової карти.

##### Призначення
Оновлює спеціальну кольорову карту з заданим індексом та параметрами.

##### Параметри
- **BaseColors**: Вказівник на базові кольори.
- **index**: Індекс спеціальної кольорової карти.
- **r1, g1, b1**: Початкові значення для колоризації.
- **r2, g2, b2**: Кінцеві значення для колоризації.

#### ReadPalette
Функція для читання палітри з lump.

##### Призначення
Читає палітру з lump з заданим номером.

##### Параметри
- **lumpnum**: Номер lump.
- **buffer**: Вказівник на буфер для зберігання палітри.

### Перелік EColorManipulation
Перелік для маніпуляцій з кольорами.

#### Призначення
Визначає різні режими маніпуляцій з кольорами.

#### Значення
- **CM_PLAIN2D**: Звичайне 2D-малювання.
- **CM_INVALID**: Недійсний режим.
- **CM_DEFAULT**: Режим за замовчуванням (без перетворення).
- **CM_FIRSTSPECIALCOLORMAP**: Перша спеціальна фіксована кольорова карта.

#### CM_MAXCOLORMAP
Макрос для визначення максимального індексу кольорової карти.

##### Призначення
Визначає максимальний індекс кольорової карти як індекс першої спеціальної кольорової карти плюс кількість спеціальних кольорових карт.

## Зв'язки з іншими модулями
Функції та структури, визначені в цьому файлі, використовуються багатьма модулями для роботи з кольорами, палітрами та кольоровими картами. Вони тісно пов'язані з системою рендерингу, палітрами та відображенням графіки.
